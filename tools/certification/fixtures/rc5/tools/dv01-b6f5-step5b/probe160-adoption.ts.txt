// B6f probe (appended in memory to src/hooks/useTimelinePlayback.test.ts; module-scope helpers are the file's own).
// Product shape of the browser build (Timeline.tsx:167, :763-781, :855-859): requiresNativeAudio false, native options
// with `enabled` (default true, app-preferences-core.ts:88) and no desktop bridge, portable options set, getTracks set.
// The rule (PARENT-NOTE-USER-CONFIRMATION-PORTABLE-LIVE-2026-11-08): every paused or stopped route ends with one live,
// prepared, paused portable session at the transport playhead, compiled from the newest tracks, with no running
// transport after Stop; a locate never cancels Play.
type B6fNode = { index: number; createdSeq: number; closed: boolean; graphs: string[]; positions: { seq: number; sec: number; running: boolean; via: string }[] }
type B6fGate = { armed: boolean; held: (() => void)[] }
const b6fGate = (): B6fGate => ({ armed: false, held: [] })
const b6fRelease = (gate: B6fGate) => {
  gate.armed = false
  for (const resume of gate.held.splice(0)) resume()
}
const b6fEditedTracks: Track[] = [track, { id: 'track-f1', name: 'B6f edit', volume: 1, clips: [] }]
let b6fSeq = 0

const withB6fWorklet = async (run: (nodes: B6fNode[], transportGate: B6fGate) => Promise<void>) => {
  const previous = {
    AudioContext: Object.getOwnPropertyDescriptor(globalThis, 'AudioContext'),
    AudioWorkletNode: Object.getOwnPropertyDescriptor(globalThis, 'AudioWorkletNode'),
    fetch: Object.getOwnPropertyDescriptor(globalThis, 'fetch'),
  }
  const nodes: B6fNode[] = []
  const transportGate = b6fGate()
  class FakeAudioContext {
    readonly sampleRate = 48_000
    readonly destination = {}
    readonly audioWorklet = { addModule: async () => undefined }
  }
  class FakeAudioWorkletNode {
    readonly port: { onmessage: ((event: { data: PortableWasmStatusMessage }) => void) | null; postMessage: (message: PortableWasmControlMessage) => void; close: () => void }
    constructor() {
      const record: B6fNode = { index: nodes.length, createdSeq: ++b6fSeq, closed: false, graphs: [], positions: [] }
      nodes.push(record)
      this.port = {
        onmessage: null,
        postMessage: (message) => {
          b6fSeq += 1
          if (message.type === 'prepare-graph') record.graphs.push(JSON.stringify(message.snapshot))
          if (message.type === 'install-schedule') record.positions.push({ seq: b6fSeq, sec: message.schedule.timeOrigin.timelineSec, running: false, via: 'schedule' })
          if (message.type === 'transport') record.positions.push({ seq: b6fSeq, sec: message.frame / 48_000, running: message.running, via: 'transport' })
          const response = portableResponseFor(message)
          if (!response) return
          const deliver = () => queueMicrotask(() => this.port.onmessage?.({ data: response }))
          if (message.type === 'transport' && message.running && transportGate.armed) transportGate.held.push(deliver)
          else deliver()
        },
        close: () => { record.closed = true },
      }
    }
    onprocessorerror: (() => {}) | null = null
    connect = () => {}
    disconnect = () => {}
  }
  const manifest = await readFile(new URL("../../public/audio-core/daw-audio-core.manifest.json", import.meta.url), "utf8")
  const wasm = await readFile(new URL("../../public/audio-core/daw-audio-core.wasm", import.meta.url))
  Object.defineProperty(globalThis, "AudioContext", { configurable: true, value: FakeAudioContext })
  Object.defineProperty(globalThis, "AudioWorkletNode", { configurable: true, value: FakeAudioWorkletNode })
  Object.defineProperty(globalThis, "fetch", { configurable: true, value: async (input: string | URL) => (
    String(input).endsWith(".manifest.json") ? new Response(manifest, { status: 200 }) : new Response(wasm, { status: 200 })) })
  try {
    await run(nodes, transportGate)
  } finally {
    for (const [key, descriptor] of Object.entries(previous)) {
      if (descriptor) Object.defineProperty(globalThis, key, descriptor)
      else Object.defineProperty(globalThis, key, { configurable: true, value: undefined })
    }
  }
}

type B6fContext = Awaited<ReturnType<typeof mountB6f>>
const mountB6f = (nodes: B6fNode[], transportGate: B6fGate, options: { nativeEnabled?: boolean; lifecycleBridge?: ReturnType<typeof createNativeHookBridge> } = {}) => {
  const fake = createFakeEngine({ clipId: 'clip-1', startSec: 1, endSec: 2 })
  const audioContext = new AudioContext()
  const resumeGate = b6fGate()
  const compileGate = b6fGate()
  const [tracks, setTracks] = createSignal<Track[]>([track])
  const [nativeEnabled, setNativeEnabled] = createSignal(options.nativeEnabled ?? true)
  const [projectId, setProjectId] = createSignal('project-a')
  const [generation, setGeneration] = createSignal(1)
  const faults: string[] = []
  const compiles: { seq: number; state: string; playheadSec: number; trackIds: string[]; volumes: number[] }[] = []
  const compileSnapshot = async (transport: Parameters<typeof compileLivePlaybackSnapshot>[0]['transport']) => {
    if (compileGate.armed) await new Promise<void>((resolve) => compileGate.held.push(resolve))
    compiles.push({ seq: ++b6fSeq, state: transport.state, playheadSec: transport.playheadSec, trackIds: tracks().map((entry) => entry.id), volumes: tracks().map((entry) => entry.volume) })
    return compileLivePlaybackSnapshot({ revision: 1, bpm: 120, transport, tracks: tracks(),
      renderState: { fx: { masterVolume: 1, masterFxInstances: [], trackFx: {} }, automationEnvelopes: [] }, sidechainRoutes: [] })
  }
  // AudioEngine.getAudioContext is null until ensureAudio creates the context (audio-engine.ts:476-478, :607-613).
  let audioCreated = false
  const engine = {
    ...fake.engine,
    ensureAudio: () => { audioCreated = true },
    getAudioContext: () => audioCreated ? audioContext : null,
    resume: async () => { if (resumeGate.armed) await new Promise<void>((resolve) => resumeGate.held.push(resolve)) },
  }
  const playback = useTimelinePlayback(engine, { getTracks: () => tracks() }, {
    enabled: nativeEnabled,
    projectId,
    projectGeneration: generation,
    compileSnapshot,
    reportFault: (fault) => faults.push('native: ' + nativePlaybackFaultText(fault)),
  }, {
    projectGeneration: generation,
    compileSnapshot,
    reportFault: (message) => faults.push('portable: ' + message),
  })
  const settle = async () => {
    let quiet = 0
    for (let turn = 0; turn < 600 && quiet < 10; turn += 1) {
      const before = b6fSeq
      await flushMicrotasks()
      await new Promise((resolve) => setTimeout(resolve, 0))
      quiet = b6fSeq === before ? quiet + 1 : 0
    }
  }
  const waitHeld = async (gate: B6fGate) => {
    for (let turn = 0; turn < 600 && gate.held.length === 0; turn += 1) {
      await flushMicrotasks()
      await new Promise((resolve) => setTimeout(resolve, 0))
    }
    expect(gate.held.length).toBeGreaterThan(0)
  }
  // Timeline.tsx:871-881 and :923-934: a structural edit rebuilds a prepared backend with the captured intent.
  const timelineReschedule = () => {
    if (!playback.isNativePlaybackPrepared() && !playback.isPortableBrowserPlaybackPrepared()) return Promise.resolve()
    return playback.restartTimelineSchedule(tracks(), {
      rebuildBackend: true,
      resumePlayback: playback.isPlaying(),
      playheadSec: playback.playheadSec(),
      owner: playback.isNativePlaybackPrepared() ? 'native' : playback.isPortableBrowserPlaybackPrepared() ? 'portable-browser' : undefined,
      projectId: projectId(),
      projectGeneration: generation(),
    })
  }
  const live = () => nodes.filter((node) => !node.closed)
  const summary = (route: string, mark: number) => {
    const open = live()
    const session = open.at(-1)
    return { route, faults: [...faults], playing: playback.isPlaying(), portableActive: playback.isPortableBrowserPlayback(),
      portablePrepared: playback.isPortableBrowserPlaybackPrepared(), nativePrepared: playback.isNativePlaybackPrepared(),
      playhead: playback.playheadSec(), openSessions: open.length, sessionPosition: session?.positions.at(-1),
      sessionTracks: session ? (session.graphs.at(-1)?.includes('"track-f1"') ? 'edited' : 'original') : undefined,
      sessionCreatedSeq: session?.createdSeq,
      sessionGain: session ? JSON.parse(session.graphs.at(-1) ?? '{}').nodes?.find((node: { id: string }) => node.id === 'track-1')?.mixer?.gain : undefined,
      runningAfterMark: nodes.flatMap((node) => node.positions.filter((position) => position.seq > mark && position.running)).length,
      compilesAfterMark: compiles.filter((entry) => entry.seq > mark), nodes: nodes.length }
  }
  const expectLivePaused = (route: string, mark: number, expectedSec: number, edited: boolean) => {
    const result = summary(route, mark)
    console.log('B6FPROBE-RESULT', JSON.stringify(result))
    expect(result.faults).toEqual([])
    expect(result.playing).toBeFalse()
    expect(result.portableActive).toBeFalse()
    expect(result.runningAfterMark).toBe(0)
    expect(result.playhead).toBeCloseTo(expectedSec, 6)
    expect(result.portablePrepared).toBeTrue()
    expect(result.openSessions).toBe(1)
    expect(result.sessionPosition?.running).toBeFalse()
    expect(result.sessionPosition?.sec ?? Number.NaN).toBeCloseTo(expectedSec, 6)
    expect(result.sessionTracks).toBe(edited ? 'edited' : 'original')
  }
  const expectLivePlaying = (route: string, mark: number, expectedSec: number, edited: boolean) => {
    const result = summary(route, mark)
    console.log('B6FPROBE-RESULT', JSON.stringify(result))
    expect(result.faults).toEqual([])
    expect(result.playing).toBeTrue()
    expect(result.portableActive).toBeTrue()
    expect(result.openSessions).toBe(1)
    expect(result.sessionPosition?.running).toBeTrue()
    expect(result.sessionPosition?.sec ?? Number.NaN).toBeCloseTo(expectedSec, 6)
    expect(result.sessionTracks).toBe(edited ? 'edited' : 'original')
  }
  const playFrom = async (sec: number) => {
    playback.setPlayhead(sec, tracks())
    await settle()
    await playback.handlePlay(tracks())
    await settle()
    expect(playback.isPlaying()).toBeTrue()
    expect(playback.isPortableBrowserPlayback()).toBeTrue()
  }
  const pausedAt = async (sec: number) => {
    await playFrom(sec)
    await playback.handlePause()
    await settle()
    expect(playback.isPortableBrowserPlaybackPrepared()).toBeTrue()
  }
  return { playback, fake, tracks, setTracks, setNativeEnabled, setProjectId, setGeneration, generation, faults, compiles,
    resumeGate, compileGate, transportGate, settle, waitHeld, timelineReschedule, summary, expectLivePaused, expectLivePlaying,
    playFrom, pausedAt, mark: () => b6fSeq }
}

const runB6f = async (body: (ctx: B6fContext) => Promise<void>, options: { nativeEnabled?: boolean; lifecycle?: boolean } = {}) => {
  const previousWindow = Object.getOwnPropertyDescriptor(globalThis, 'window')
  const lifecycleBridge = options.lifecycle ? createNativeHookBridge(undefined, {
    initial: { state: 'ready', powerGeneration: 1 },
    completeRecovery: async () => ({ accepted: true }),
    retryRecovery: async () => ({ accepted: true }),
  }) : undefined
  Object.defineProperty(globalThis, 'window', { configurable: true,
    value: lifecycleBridge ? { dawDesktop: { audioHost: lifecycleBridge.audioHost } } : {} })
  try {
    await withB6fWorklet(async (nodes, transportGate) => {
      await withFakeRaf(async () => {
        await createRoot(async (dispose) => {
          const ctx = mountB6f(nodes, transportGate, { nativeEnabled: options.nativeEnabled })
          await ctx.settle()
          try {
            await body(Object.assign(ctx, { lifecycleBridge }))
          } finally {
            dispose()
          }
        })
      })
    })
  } finally {
    if (previousWindow) Object.defineProperty(globalThis, 'window', previousWindow)
    else Reflect.deleteProperty(globalThis, 'window')
  }
}

// The B6d8/F1/F1b analog: an edit while playing and, in the same tick, the Timeline's resuming rebuild. The action
// lands `delay` microtask turns later, or while the rebuild's Play is held at its portable compile or transport start.
const b6fRebuildRoute = async (ctx: B6fContext, action: 'pause' | 'stop' | 'locate', delay: number, gate?: 'compile' | 'transport') => {
  await ctx.playFrom(1)
  ctx.setTracks(b6fEditedTracks)
  const held = gate === 'compile' ? ctx.compileGate : gate === 'transport' ? ctx.transportGate : undefined
  if (held) held.armed = true
  const rebuild = ctx.playback.restartTimelineSchedule(ctx.tracks(), { rebuildBackend: true, resumePlayback: true, playheadSec: 2,
    owner: 'portable-browser', projectId: 'project-a', projectGeneration: ctx.generation() })
  if (held) await ctx.waitHeld(held)
  else for (let turn = 0; turn < delay; turn += 1) await Promise.resolve()
  const mark = ctx.mark()
  let action_: Promise<unknown> = Promise.resolve()
  if (action === 'pause') action_ = ctx.playback.handlePause()
  if (action === 'stop') action_ = ctx.playback.handleStop()
  if (action === 'locate') ctx.playback.setPlayhead(3, ctx.tracks())
  const playheadAtAction = ctx.playback.playheadSec()
  await ctx.settle()
  if (held) b6fRelease(held)
  await Promise.allSettled([rebuild, action_])
  await ctx.settle()
  const name = `rebuild ${action} ${gate ?? 'delay=' + delay}`
  if (action === 'locate') ctx.expectLivePlaying(name, mark, 3, true)
  else ctx.expectLivePaused(name, mark, action === 'stop' ? 0 : playheadAtAction, true)
}
for (const action of ['pause', 'stop', 'locate'] as const) {
  for (let delay = 0; delay <= 40; delay += 1)
    test(`B6FPROBE rebuild ${action} delay=${delay}`, async () => {
      await runB6f((ctx) => b6fRebuildRoute(ctx, action, delay))
    })
  for (const gate of ['compile', 'transport'] as const)
    test(`B6FPROBE rebuild ${action} gate=${gate}`, async () => {
      await runB6f((ctx) => b6fRebuildRoute(ctx, action, 0, gate))
    })
}

        await Promise.allSettled([play, cancel])
        await b6f3Expect(live, mark, b6f3Live(action === 'stop' ? 0 : 1, 'none'), 'cancel-pending-play')
      }, ne)
    })
